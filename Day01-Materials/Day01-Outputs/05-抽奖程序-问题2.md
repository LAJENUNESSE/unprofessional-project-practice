# 抽奖程序-问题2

```python
import tkinter as tk
from tkinter import messagebox
import random
import time

# 30人虚拟名单
NAMES = [
    "张伟", "王芳", "李娜", "刘洋", "陈静", "杨帆", "赵磊", "黄敏", "周杰", "吴婷",
    "徐强", "孙丽", "马超", "朱霞", "胡军", "郭涛", "林梅", "何平", "高飞", "罗丹",
    "郑洁", "梁波", "谢楠", "宋佳", "唐磊", "韩雪", "冯刚", "曹颖", "许巍", "邓超"
]

root = tk.Tk()
root.title("随机抽奖")
root.geometry("300x200")
root.resizable(False, False)

result_label = tk.Label(root, text="点击下方按钮开始抽奖",
                        font=("微软雅黑", 18, "bold"), fg="#333333")
result_label.pack(expand=True)

draw_button = tk.Button(root, text="开始抽奖", font=("微软雅黑", 14),
                        width=12, bg="#4CAF50", fg="white",
                        activebackground="#45a049", relief="raised", bd=2)
draw_button.pack(pady=20)

rolling = False  # 防止重复点击


def on_draw():
    global rolling
    if rolling:
        return
    rolling = True
    draw_button.config(state="disabled", text="抽奖中...")
    end_time = time.time() + 5  # 名字滚动5秒

    def update():
        if time.time() < end_time:
            result_label.config(text=random.choice(NAMES))
            root.after(50, update)
        else:
            winner = random.choice(NAMES)  # 确定最终中奖人
            result_label.config(text=winner, fg="red")
            # 按钮背景改为蓝色
            draw_button.config(state="normal", text="开始抽奖",
                               bg="blue", activebackground="blue")
            rolling = False
            messagebox.showinfo("中奖结果", f"🎉 恭喜「{winner}」中奖！")

    update()


draw_button.config(command=on_draw)
root.mainloop()
```
