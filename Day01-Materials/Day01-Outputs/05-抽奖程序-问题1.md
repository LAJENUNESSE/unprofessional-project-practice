# 抽奖程序-问题1

```python
import tkinter as tk
import random

# 30人虚拟名单
names = [
    "张伟", "王芳", "李娜", "刘强", "陈静",
    "杨洋", "赵敏", "黄磊", "周杰", "吴倩",
    "徐峥", "孙丽", "马超", "朱婷", "胡军",
    "郭涛", "林峰", "何炅", "高翔", "罗兰",
    "郑爽", "梁波", "宋佳", "唐嫣", "许晴",
    "韩雪", "冯刚", "邓超", "曹颖", "彭飞"
]

rolling = False  # 是否正在滚动


def roll(count=20):
    """滚动效果，逐渐减速后揭晓中奖者"""
    global rolling
    if count > 0:
        result_label.config(text=random.choice(names), fg="orange")
        delay = 30 + (20 - count) * 8  # 逐渐变慢
        root.after(delay, roll, count - 1)
    else:
        winner = random.choice(names)
        result_label.config(text=f"中奖者：{winner}", fg="red")
        rolling = False
        draw_button.config(state="normal", text="开始抽奖")


def start_draw():
    global rolling
    if rolling:
        return
    rolling = True
    draw_button.config(state="disabled", text="抽奖中...")
    roll(20)


root = tk.Tk()
root.title("随机抽奖")
root.geometry("300x200")
root.resizable(False, False)

title_label = tk.Label(root, text="🎉 幸运大抽奖 🎉",
                       font=("Microsoft YaHei", 15, "bold"))
title_label.pack(pady=12)

result_label = tk.Label(root, text="点击下方按钮开始",
                        font=("Microsoft YaHei", 14), fg="blue")
result_label.pack(pady=10)

draw_button = tk.Button(root, text="开始抽奖",
                        font=("Microsoft YaHei", 12),
                        command=start_draw,
                        bg="#4CAF50", fg="white",
                        width=10, height=1,
                        relief="raised", cursor="hand2")
draw_button.pack(pady=10)

root.mainloop()
```
